# download the training set 


mkdir imagenet && cd imagenet

wget -c https://image-net.org/data/ILSVRC/2012/ILSVRC2012_img_train.tar


mkdir train
tar -xf ILSVRC2012_img_train.tar -C train
rm ILSVRC2012_img_train.tar

cd train
for f in *.tar; do
    mkdir "${f%.tar}"
    tar -xf "$f" -C "${f%.tar}"
    rm "$f"
done
